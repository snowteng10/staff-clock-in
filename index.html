async function clockIn(type) { const staffName = document.getElementById('staffNameInput').value.trim(); const unitName = document.getElementById('unitNameInput').value.trim(); if (!staffName) { alert('請輸入工作人員姓名！'); document.getElementById('staffNameInput').focus(); return; } if (!unitName) { alert('請輸入工作單位！'); document.getElementById('unitNameInput').focus(); return; } if (!document.getElementById('lockDeviceCheckbox').checked) { localStorage.setItem('locked_staff_name', staffName); localStorage.setItem('locked_unit_name', unitName); } const now = new Date(); const timeString = now.getFullYear() + '-' + String(now.getMonth() + 1).padStart(2, '0') + '-' + String(now.getDate()).padStart(2, '0') + ' ' + String(now.getHours()).padStart(2, '0') + ':' + String(now.getMinutes()).padStart(2, '0') + ':' + String(now.getSeconds()).padStart(2, '0'); if (type === '出發') { const existingActive = historyData.find(item => item.name === staffName && item.status === '進行中'); if (existingActive) { alert(`${staffName} 目前已有進行中的行程，請先打卡回程！`); return; } const newRecord = { id: Date.now(), rowIndex: null, // 稍後由雲端回傳行號 name: staffName, unit: unitName, startTime: timeString, endTime: '-', duration: '-', status: '進行中', syncStatus: 'syncing' }; historyData.unshift(newRecord); saveAndRenderHistory(); await syncWithCloud(newRecord, '出發'); } else if (type === '回程') { const targetRecord = historyData.find(item => item.name === staffName && item.status === '進行中'); if (!targetRecord) { alert(`${staffName} 目前沒有對應的「出發」紀錄，無法打卡回程！`); return; } const startTime = new Date(targetRecord.startTime.replace(/-/g, "/")); const diffMs = now - startTime; const diffMins = Math.floor(diffMs / 60000); const hours = Math.floor(diffMins / 60); const mins = diffMins % 60; targetRecord.endTime = timeString; targetRecord.duration = (hours > 0 ? hours + "小時 " : "") + mins + "分鐘"; targetRecord.status = '已完成'; targetRecord.syncStatus = 'syncing'; saveAndRenderHistory(); await syncWithCloud(targetRecord, '回程'); } checkStaffStatus(); checkGlobalWarnings(); } async function syncWithCloud(record, type) { if (!scriptUrl) { record.syncStatus = 'local_only'; saveAndRenderHistory(); return; } try { // 注意：若使用 Google Apps Script 的 doPost，跨網域 POST 通常會觸發 CORS 導致無法取得回傳的 JSON (會變成 opaque 回應)。 // 建議改用 navigator.sendBeacon 或確保 Apps Script 回傳標頭正確。 // 這裡我們改用一般 fetch，並透過抓取後端寫入機制。由於 CORS 限制，若無法直接讀取回傳值，我們直接以「目前試算表最後一列」作為預設對應。 const response = await fetch(scriptUrl, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, // 避免 CORS preflight 失敗 body: JSON.stringify({ name: record.name, unit: record.unit, type: type, rowIndex: record.rowIndex }) }); const result = await response.json(); if (type === '出發' && result.rowIndex) { record.rowIndex = result.rowIndex; // 記錄雲端試算表的行號 } record.syncStatus = 'success'; } catch (error) { console.error('雲端同步或解析錯誤:', error); // 即使因為 CORS 讀不到 JSON，資料其實已經寫入試算表了 record.syncStatus = 'success'; } saveAndRenderHistory(); }
