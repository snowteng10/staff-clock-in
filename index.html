function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var now = new Date(); var dateStr = Utilities.formatDate(now, "GMT+8", "yyyy-MM-dd"); var timeStr = Utilities.formatDate(now, "GMT+8", "yyyy-MM-dd HH:mm:ss"); if (data.type === '出發') { // 依序寫入：日期、姓名、組別、出發時間、回程時間(空)、累計時間(空) sheet.appendRow([dateStr, data.name, data.unit, timeStr, "", ""]); } else if (data.type === '回程') { var rows = sheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { // 檢查姓名(第2欄)與組別(第3欄)，且回程時間(第5欄)為空 if (rows[i][1] === data.name && rows[i][2] === data.unit && rows[i][4] === "") { var departureTimeStr = rows[i][3]; var departureDate = new Date(departureTimeStr.replace(/-/g, "/")); var returnDate = now; var diffMs = returnDate - departureDate; var diffMins = Math.round(diffMs / 60000); // 更新第5欄(E欄：回程時間)與第6欄(F欄：累計分鐘) sheet.getRange(i + 1, 5).setValue(timeStr); sheet.getRange(i + 1, 6).setValue(diffMins); break; } } } return ContentService.createTextOutput(JSON.stringify({"status": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
